<script>
// FIX v5b: Audio module (engine synth, SFX, skid, limiter chop)
const FUNDAMENTAL_MULT = 1;   // スマホのスピーカーで低音が薄い時は 2 にする（1オクターブ上げる）
const MASTER_VOLUME = 0.5;
const SFX_MIN_INTERVAL = 80;  // ms（同じ名前の効果音の連発防止）

var audioCtx = null, masterGain = null; // var: updateUI からの参照でTDZエラーを避ける
let soundMuted = false;                 // FIX v5b: 起動時はミュートしない（最初に触れば鳴る）
let osc1, osc2, osc3, filter, engineGain, noiseBuffer, noiseBandpass, noiseGain, skidBandpass, skidGain;
const lastSfxTimes = {};                // FIX v5b: 効果音の名前ごとに最終時刻を持つ
let prevIsAccelOn = false, prevRPM = 900;

function initAudio() {
  if (audioCtx) {
    if (audioCtx.state !== 'running') audioCtx.resume().catch(() => {});
    return;
  }
  try {
    const AC = window.AudioContext || window.webkitAudioContext;
    audioCtx = new AC();
    masterGain = audioCtx.createGain();
    masterGain.gain.value = soundMuted ? 0 : MASTER_VOLUME;
    masterGain.connect(audioCtx.destination);

    // 共有ノイズバッファ（約2秒）。吸気音・スキール・バーストで使い回す
    const len = Math.floor(audioCtx.sampleRate * 2);
    noiseBuffer = audioCtx.createBuffer(1, len, audioCtx.sampleRate);
    const d = noiseBuffer.getChannelData(0);
    for (let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1;

    // --- Engine ---
    engineGain = audioCtx.createGain();
    engineGain.gain.value = 0.2;
    filter = audioCtx.createBiquadFilter();
    filter.type = 'lowpass';
    filter.frequency.value = 800;
    filter.Q.value = 1.2;
    engineGain.connect(filter);
    filter.connect(masterGain);

    osc1 = audioCtx.createOscillator();
    osc1.type = 'sawtooth';
    osc1.frequency.value = 30;
    osc1.connect(engineGain);
    osc1.start();

    osc2 = audioCtx.createOscillator();
    osc2.type = 'sawtooth';
    osc2.frequency.value = 60;
    osc2.detune.value = 6;
    const g2 = audioCtx.createGain();
    g2.gain.value = 0.5;
    osc2.connect(g2);
    g2.connect(engineGain);
    osc2.start();

    osc3 = audioCtx.createOscillator();
    osc3.type = 'square';
    osc3.frequency.value = 15;
    const g3 = audioCtx.createGain();
    g3.gain.value = 0.25;
    osc3.connect(g3);
    g3.connect(engineGain);
    osc3.start();

    // --- Intake noise ---
    const nSrc = audioCtx.createBufferSource();
    nSrc.buffer = noiseBuffer;
    nSrc.loop = true;
    noiseBandpass = audioCtx.createBiquadFilter();
    noiseBandpass.type = 'bandpass';
    noiseBandpass.frequency.value = 1200;
    noiseBandpass.Q.value = 2.0;
    noiseGain = audioCtx.createGain();
    noiseGain.gain.value = 0.02;
    nSrc.connect(noiseBandpass);
    noiseBandpass.connect(noiseGain);
    noiseGain.connect(masterGain);
    nSrc.start();

    // --- Tire skid ---
    const sSrc = audioCtx.createBufferSource();
    sSrc.buffer = noiseBuffer;
    sSrc.loop = true;
    skidBandpass = audioCtx.createBiquadFilter();
    skidBandpass.type = 'bandpass';
    skidBandpass.frequency.value = 1800;
    skidBandpass.Q.value = 5.0;
    skidGain = audioCtx.createGain();
    skidGain.gain.value = 0;
    sSrc.connect(skidBandpass);
    skidBandpass.connect(skidGain);
    skidGain.connect(masterGain);
    sSrc.start();

    if (audioCtx.state !== 'running') audioCtx.resume().catch(() => {});
  } catch (e) {
    console.error('Web Audio init failed', e);
    audioCtx = null;
    masterGain = null;
  }
}

function updateAudio(displayRPM, accel, isAccelOn, limiterActive, sidebrake, speed) {
  // FIX v5b: 前フレームの値は早期returnの前に更新する（ミュート解除直後の空ブローオフ防止）
  const wasOn = prevIsAccelOn, wasRPM = prevRPM;
  prevIsAccelOn = isAccelOn;
  prevRPM = displayRPM;
  if (!audioCtx || audioCtx.state !== 'running' || soundMuted) return;

  const now = audioCtx.currentTime;
  const f = (displayRPM / 30.0) * FUNDAMENTAL_MULT;
  osc1.frequency.setTargetAtTime(f, now, 0.04);
  osc2.frequency.setTargetAtTime(f * 2, now, 0.04);
  osc3.frequency.setTargetAtTime(f * 0.5, now, 0.04);

  filter.frequency.setTargetAtTime(400 + displayRPM * 0.5 + (isAccelOn ? accel * 1500 : 0), now, 0.04);
  noiseBandpass.frequency.setTargetAtTime(800 + (displayRPM / 7500) * 1700, now, 0.04);
  noiseGain.gain.setTargetAtTime(isAccelOn ? accel * 0.08 : 0.01, now, 0.04);

  // FIX v5b: リミッター断続は20Hz（周期50ms）、時定数も短くして聞こえるようにする
  let limiterMult = 1.0, engineTC = 0.03;
  if (limiterActive) {
    limiterMult = (Math.floor(now * 40) % 2 === 0) ? 0.3 : 1.0;
    engineTC = 0.005;
  }
  engineGain.gain.setTargetAtTime((isAccelOn ? (0.25 + accel * 0.35) : 0.18) * limiterMult, now, engineTC);

  // アクセルを離した瞬間：ブローオフ / バックファイア
  if (wasOn && !isAccelOn) {
    if (wasRPM > 4500) triggerNoiseBurst(2500, 0.15, 0.25);
    if (wasRPM > 6000) {
      const pops = Math.floor(Math.random() * 3); // 0〜2回
      let delay = 0;
      for (let i = 0; i < pops; i++) {
        delay += 50 + Math.random() * 150;
        const dur = 0.04 + Math.random() * 0.04;
        setTimeout(() => triggerNoiseBurst(1000 + Math.random() * 400, dur, 0.3), delay);
      }
    }
  }

  // サイドブレーキのスキール音
  if (sidebrake && speed > 4.0) skidGain.gain.setTargetAtTime(Math.min(0.3, speed * 0.012), now, 0.05);
  else skidGain.gain.setTargetAtTime(0, now, 0.05);
}

function triggerNoiseBurst(centerFreq, duration, volume) {
  if (!audioCtx || audioCtx.state !== 'running' || soundMuted || !noiseBuffer) return;
  try {
    const now = audioCtx.currentTime;
    const src = audioCtx.createBufferSource();
    src.buffer = noiseBuffer;
    const bp = audioCtx.createBiquadFilter();
    bp.type = 'bandpass';
    bp.frequency.value = centerFreq;
    bp.Q.value = 3.0;
    const g = audioCtx.createGain();
    g.gain.setValueAtTime(volume, now);
    g.gain.exponentialRampToValueAtTime(0.001, now + duration);
    src.connect(bp);
    bp.connect(g);
    g.connect(masterGain);
    src.start(now, Math.random() * 1.5, duration);
  } catch (e) {}
}

function playClick() {
  const t = audioCtx.currentTime;
  const o = audioCtx.createOscillator(), g = audioCtx.createGain();
  o.type = 'square';
  o.frequency.setValueAtTime(180, t);
  o.frequency.exponentialRampToValueAtTime(90, t + 0.04);
  g.gain.setValueAtTime(0.3, t);
  g.gain.exponentialRampToValueAtTime(0.001, t + 0.04);
  o.connect(g); g.connect(masterGain);
  o.start(t); o.stop(t + 0.04);
}

function playBlip() {
  const t = audioCtx.currentTime;
  const o = audioCtx.createOscillator(), g = audioCtx.createGain();
  o.type = 'sine';
  o.frequency.setValueAtTime(1200, t);
  g.gain.setValueAtTime(0.1, t);
  g.gain.exponentialRampToValueAtTime(0.001, t + 0.03);
  o.connect(g); g.connect(masterGain);
  o.start(t); o.stop(t + 0.03);
}

function triggerGrindSFX(freq, duration, volume) {
  const t = audioCtx.currentTime;
  const o = audioCtx.createOscillator(), g = audioCtx.createGain();
  o.type = 'sawtooth';
  o.frequency.setValueAtTime(freq, t);
  g.gain.setValueAtTime(volume, t);
  g.gain.exponentialRampToValueAtTime(0.001, t + duration);
  o.connect(g); g.connect(masterGain);
  o.start(t); o.stop(t + duration);
  triggerNoiseBurst(600, duration, volume * 0.8);
}

// SMOOTH は高いブリップだけ（クリックは GEAR_CHANGE 側が鳴らすので二重にしない）
function playSfx(name) {
  if (!audioCtx || audioCtx.state !== 'running' || soundMuted) return;
  const t = performance.now();
  if (lastSfxTimes[name] !== undefined && t - lastSfxTimes[name] < SFX_MIN_INTERVAL) return;
  lastSfxTimes[name] = t;
  try {
    if (name === 'GEAR_CHANGE') playClick();
    else if (name === 'SMOOTH') playBlip();
    else if (name === 'SHOCK_SMALL') triggerGrindSFX(80, 0.1, 0.35);
    else if (name === 'SHOCK_LARGE' || name === 'LAG' || name === 'OVERREV') triggerGrindSFX(60, 0.22, 0.5);
  } catch (e) {}
}

const PX = new Float32Array(DRAW_DISTANCE + 1), PY = new Float32Array(DRAW_DISTANCE + 1);

function quad(a, b, c, d, e, f, g, h) {

function render(curCurve) {
  const W = canvas.width, H = canvas.height, half = W / 2, horizon = H * HORIZON;
  const cd = 1 / Math.tan(FOV_DEG * Math.PI / 360);

  ctx.save();
  ctx.translate(W / 2, H * 0.75); ctx.rotate(roll); ctx.translate(-W / 2, -H * 0.75);

  sky.addColorStop(0, '#0b1a3a'); sky.addColorStop(1, '#5a7fb0');
  ctx.fillStyle = sky; ctx.fillRect(-W, -H, 3 * W, horizon + H);

  // 山（カーブとステアで横に流れる。距離では流れない）
  ctx.fillStyle = '#2a3552';
  ctx.beginPath(); ctx.moveTo(-W, horizon);
  const off = ((bgX % 150) + 150) % 150, shift = Math.floor(bgX / 150);
  for (let i = -3; i < Math.ceil(W / 150) + 3; i++) {
    const k = i - shift, hgt = 22 + (((k * 37) % 5 + 5) % 5) * 9;
    const x0 = i * 150 + off;
    ctx.lineTo(x0 + 75, horizon - hgt); ctx.lineTo(x0 + 150, horizon);
  }
  ctx.lineTo(2 * W, horizon); ctx.closePath(); ctx.fill();

  // 地面のベース

  const baseSeg = Math.floor(trackZ / SEG_LEN), bp = trackZ / SEG_LEN - baseSeg;
  const K = KAPPA * SEG_LEN * SEG_LEN;
  let x = 0, dx = -segs[baseSeg % N] * bp * K;

    const sc = cd / z;
    PS[k] = sc;
    PX[k] = half + sc * (x - playerX * ROAD_HALF) * half;
    PY[k] = horizon + sc * CAMERA_HEIGHT * half;
    PW[k] = sc * ROAD_HALF * half;

  }

  for (let n = DRAW_DISTANCE - 1; n >= 0; n--) {
    const y1 = PY[n], y2 = PY[n + 1];
    if (y2 >= y1) continue;
    const idx = (baseSeg + n) % N, alt = (Math.floor(idx / 3) % 2) === 0;

// ===== メインループ =====
let last = performance.now();
function loop(ts) {
  let dt = (ts - last) / 1000; last = ts; dt = Math.min(dt, 0.05);

  // ジャイロが来ていなければキーボード
  if (performance.now() - gyroLastTime > 1500) {
    gyroActive = false;
    steerRaw = ((keys['arrowright'] || keys['d']) ? 1 : 0) - ((keys['arrowleft'] || keys['a']) ? 1 : 0);
  }
  steerFiltered += (steerRaw - steerFiltered) * STEER_SMOOTH;

  // 横方向の車両ダイナミクス（speed, sidebrake は既存コードのグローバル）
  const sp = speed / SPEED_REF, drift = sidebrake;
  playerX += steerFiltered * sp * dt * STEER_GAIN * (drift ? DRIFT_STEER_MULT : 1);
  const curCurve = segs[Math.floor(trackZ / SEG_LEN) % N];
  playerX -= sp * sp * curCurve * dt * CENTRIFUGAL * (drift ? DRIFT_CENTRI_MULT : 1);

  trackZ += speed * dt;
  if (trackZ >= trackLen) trackZ -= trackLen;
  if (trackZ < 0) trackZ += trackLen;

  // オフロードと壁
  if (Math.abs(playerX) > 1.0) speed *= (1 - 0.9 * dt);
  wallCooldown = Math.max(0, wallCooldown - dt);
  if (Math.abs(playerX) > WALL_X) {
    playerX = playerX > 0 ? 2.0 : -2.0;
    if (wallCooldown === 0) {
      speed *= 0.5; wallCooldown = 0.5; flash = 1;
      if (typeof playSfx === 'function') playSfx('SHOCK_LARGE');
    }
  }
  flash = Math.max(0, flash - dt * 2.5);

  // 画面ロールと山のスクロール
  const rollTarget = Math.max(-MAX_ROLL, Math.min(MAX_ROLL,
    (steerFiltered * 0.6 + curCurve * 0.12) * Math.min(sp, 1) * (drift ? 1.5 : 1) * MAX_ROLL));
  roll += (rollTarget - roll) * Math.min(1, dt * 6);
  bgX -= (curCurve * sp * 40 + steerFiltered * sp * 10) * dt;

  render(curCurve);

  uiTimer += dt;
  if (uiTimer >= 0.1) {
    uiTimer = 0;
    extraDebug.textContent = 'STEER ' + (steerFiltered * STEER_MAX_DEG).toFixed(1) + '° (' + steerFiltered.toFixed(2) + ') | X ' + playerX.toFixed(2) + ' | LAP ' + (trackZ / trackLen * 100).toFixed(1) + '%';
  }
